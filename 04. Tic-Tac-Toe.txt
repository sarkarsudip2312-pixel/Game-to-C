char board[3][3] = {
{'1', '2', '3'},
{'4', '5', '6'},
{'7', '8', '9'}
};

void displayBoard() {
printf("\n");
printf("     |     |     \n");
printf("  %c  |  %c  |  %c  \n", board[0][0], board[0][1], board[0][2]);
printf("*****|*****|*****\n");
printf("     |     |     \n");
printf("  %c  |  %c  |  %c  \n", board[1][0], board[1][1], board[1][2]);
printf("*****|*****|*****\n");
printf("     |     |     \n");
printf("  %c  |  %c  |  %c  \n", board[2][0], board[2][1], board[2][2]);
printf("     |     |     \n\n");
}

int main() {
int player = 1;
int choice, row, col, moves = 0;
char mark;

```
printf("================================\n");
printf("       TIC-TAC-TOE GAME\n");
printf("================================\n");

printf("Player 1: X\n");
printf("Player 2: O\n");

while (moves < 9) {
    displayBoard();

    mark = (player == 1) ? 'X' : 'O';

    printf("Player %d (%c), enter a position (1-9): ",
           player, mark);

    if (scanf("%d", &choice) != 1) {
        printf("Invalid input! Enter a number.\n");
        return 1;
    }

    if (choice < 1 || choice > 9) {
        printf("Invalid position! Choose 1-9.\n");
        continue;
    }

    row = (choice - 1) / 3;
    col = (choice - 1) % 3;

    if (board[row][col] == 'X' ||
        board[row][col] == 'O') {
        printf("This position is already occupied!\n");
        continue;
    }

    board[row][col] = mark;
    moves++;

    if (checkWin()) {
        displayBoard();
        printf("Congratulations! Player %d (%c) wins!\n",
               player, mark);
        return 0;
    }

    player = (player == 1) ? 2 : 1;
}

displayBoard();
printf("It's a Draw! Nobody wins.\n");

return 0;
```

}
