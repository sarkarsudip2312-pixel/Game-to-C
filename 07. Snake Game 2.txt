#include <stdio.h>
#include <stdlib.h>
#include <conio.h>
#include <windows.h>
#include <time.h>

#define WIDTH 30
#define HEIGHT 15

int x[100], y[100];
int length = 3;
int foodX, foodY;
int score = 0;
int gameOver = 0;
int direction = 1;

// Directions:
// 0 = Up, 1 = Right, 2 = Down, 3 = Left

void setup() {
x[0] = WIDTH / 2;
y[0] = HEIGHT / 2;

```
x[1] = x[0] - 1;
y[1] = y[0];

x[2] = x[1] - 1;
y[2] = y[1];

srand((unsigned int)time(NULL));

foodX = rand() % WIDTH;
foodY = rand() % HEIGHT;
```

}

void draw() {
system("cls");

```
printf("========== SNAKE GAME ==========\n");
printf("Score: %d\n", score);
printf("Controls: W = Up, A = Left\n");
printf("          S = Down, D = Right\n");
printf("          X = Quit\n\n");

for (int row = -1; row <= HEIGHT; row++) {
    for (int col = -1; col <= WIDTH; col++) {

        if (row == -1 || row == HEIGHT ||
            col == -1 || col == WIDTH) {
            printf("#");
        }
        else if (row == y[0] && col == x[0]) {
            printf("O");
        }
        else if (row == foodY && col == foodX) {
            printf("*");
        }
        else {
            int printed = 0;

            for (int i = 1; i < length; i++) {
                if (x[i] == col && y[i] == row) {
                    printf("o");
                    printed = 1;
                    break;
                }
            }

            if (!printed)
                printf(" ");
        }
    }
    printf("\n");
}
```

}

void input() {
if (_kbhit()) {
char key = _getch();

```
    if ((key == 'w' || key == 'W') && direction != 2)
        direction = 0;
    else if ((key == 'd' || key == 'D') && direction != 3)
        direction = 1;
    else if ((key == 's' || key == 'S') && direction != 0)
        direction = 2;
    else if ((key == 'a' || key == 'A') && direction != 1)
        direction = 3;
    else if (key == 'x' || key == 'X')
        gameOver = 1;
}
```

}

void update() {
int newX = x[0];
int newY = y[0];

```
if (direction == 0) newY--;
else if (direction == 1) newX++;
else if (direction == 2) newY++;
else if (direction == 3) newX--;

// Check wall collision
if (newX < 0 || newX >= WIDTH ||
    newY < 0 || newY >= HEIGHT) {
    gameOver = 1;
    return;
}

// Check food
int ateFood = (newX == foodX && newY == foodY);

// Check body collision
int bodyLimit = length - (ateFood ? 0 : 1);

for (int i = 0; i < bodyLimit; i++) {
    if (x[i] == newX && y[i] == newY) {
        gameOver = 1;
        return;
    }
}

// Move the snake
if (ateFood && length < 100) {
    length++;
    score += 10;
}

for (int i = length - 1; i > 0; i--) {
    x[i] = x[i - 1];
    y[i] = y[i - 1];
}

x[0] = newX;
y[0] = newY;

// Generate new food
if (ateFood) {
    int valid;

    do {
        valid = 1;
        foodX = rand() % WIDTH;
        foodY = rand() % HEIGHT;

        for (int i = 0; i < length; i++) {
            if (x[i] == foodX && y[i] == foodY) {
                valid = 0;
                break;
            }
        }
    } while (!valid);
}
```

}

int main() {
setup();

```
while (!gameOver) {
    draw();
    input();

    if (!gameOver) {
        update();
        Sleep(120);
    }
}

system("cls");

printf("\n========== GAME OVER ==========\n");
printf("Your Final Score: %d\n", score);
printf("Snake Length: %d\n", length);
printf("\nThanks for playing!\n");

printf("\nPress any key to exit...");
_getch();

return 0;
```

}
