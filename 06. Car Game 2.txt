void input() {
if (_kbhit()) {
char key = _getch();

```
    if (key == 'a'  key == 'A') {
        if (carPosition  0)
            carPosition--;
    }
    else if (key == 'd'  key == 'D') {
        if (carPosition  ROAD_WIDTH - 1)
            carPosition++;
    }
    else if (key == 'q'  key == 'Q') {
        gameOver = 1;
    }
}
```

}

void update() {
obstacleRow++;

```
if (obstacleRow == ROAD_HEIGHT - 1 &&
    obstaclePosition == carPosition) {
    gameOver = 1;
    return;
}

if (obstacleRow = ROAD_HEIGHT) {
    obstacleRow = 0;
    obstaclePosition = rand() % ROAD_WIDTH;
    score += 10;
}
```

}
